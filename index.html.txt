 $('#workrole').addEventListener('change',()=>{$('#grade').value='';choose()});$('#side-subject').addEventListener('change',choose);

}

function setTextSize(size){

 const selected=size==='large'?'large':'normal';

 document.documentElement.dataset.textSize=selected;

 document.querySelectorAll('.size-button').forEach(button=>button.setAttribute('aria-pressed',String(button.dataset.textSize===selected)));

 persist('elementary-ai-text-size',selected);

}
